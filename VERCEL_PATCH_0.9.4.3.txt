در vercel.json، قبل از wildcard عمومی این Rewrite را اضافه کنید:

{"source":"/api/v1/admin/test-reset","destination":"/api/test-reset.js"}

نمونه:
{
  "$schema":"https://openapi.vercel.sh/vercel.json",
  "rewrites":[
    {"source":"/diag/ping","destination":"/api/ping.js"},
    {"source":"/api/v1/admin/test-reset","destination":"/api/test-reset.js"},
    {"source":"/api/v1/knowledge/document-relations","destination":"/api/document-relations.js"},
    {"source":"/api/(.*)","destination":"/api/index.js"}
  ]
}
